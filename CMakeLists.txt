cmake_minimum_required(VERSION 3.16)
project(pinball_mac CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG -fno-exceptions -fno-rtti -fvisibility=hidden")
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
find_library(SDL2_LIB SDL2 PATHS /opt/homebrew/lib REQUIRED)
add_executable(pinball src/main.cpp src/partout.cpp src/physics.cpp src/audio.cpp src/music.cpp src/scene.cpp src/rules.cpp)
target_include_directories(pinball PRIVATE /opt/homebrew/include/SDL2)
find_library(AUDIOTOOLBOX AudioToolbox REQUIRED)
find_library(COREFOUNDATION CoreFoundation REQUIRED)
target_link_libraries(pinball ${SDL2_LIB} ${AUDIOTOOLBOX} ${COREFOUNDATION})
target_link_options(pinball PRIVATE -Wl,-dead_strip)
